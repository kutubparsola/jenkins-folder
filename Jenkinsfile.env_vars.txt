pipeline{
    agent any 
    environment{
        global_var = "text from global vars"
    }
    stages{
        stage('build'){
            environment{
                stage_var= "this is stage variable"
            }
            steps{
                echo "build stage"
                echo "$global_var"
                echo "$env.stage_var"
            }
        }
        stage('test'){
            steps{
                echo "test stage"
                echo '${global_var}'
            }
        }
    }
}
